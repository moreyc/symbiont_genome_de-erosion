#### Genome annotation, gene clustering, and mobile element identification ####

## Genome annotation ##

mkdir -f data/faa
mkdir -f data/ffn
mkdir -f data/gff

prokka data/$GENOME \
	--outdir prokka/$GENOME \
	--prefix $GENOME \
	--locustag $GENOME \
	--force
cp prokka/$GENOME/$GENOME.faa data/faa
cp prokka/$GENOME/$GENOME.ffn data/ffn
cp prokka/$GENOME/$GENOME.gff data/gff

## Gene clustering by orthogroup ##
orthofinder -f data/faa -t 8 -a 8 -o results/{EXAMPLE}/orthofinder/

## Insertion sequence prediction ##
isescan.py \
	--seqfile data/genomes/$GENOME.fna \
	--output insertion_sequences/isescan/$GENOME
